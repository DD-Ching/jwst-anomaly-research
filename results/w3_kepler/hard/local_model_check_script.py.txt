"""Model-only check of the local shoulder rule (PLAN section 3.1), committed after the review
(finding: not stored). Same computation as run before the plan commit (10:53:04 UTC).

    python results/w3_kepler/hard/local_model_check_script.py.txt > results/w3_kepler/hard/local_model_check.json

Noise-free, exposure-averaged W3 flux, 160 u0 values in [0, 1.995]; per cell, the share of u0
values for which the two sides do not both have >= 2 locally bright cadences (the rule's windows,
baseline median and scatter, thresholds max(0.10 b, 5 s_B); sigma = 0 here).
"""

import json
import sys
from pathlib import Path

import numpy as np

sys.path.insert(0, str(Path(__file__).resolve().parents[3] / "scripts"))
import w3_kepler as wk  # noqa: E402


def ok_side(t, f, ie, span, sgn):
    ws, wb = 0.25 * span, max(1.0, span)
    d = (t[ie] - t) if sgn < 0 else (t - t[ie])
    s, b = (d > 0) & (d <= ws), (d > ws) & (d <= ws + wb)
    base = np.median(f[b])
    s_b = 1.4826 * np.median(np.abs(f[b] - base))
    return ((f[s] - base) >= max(0.10 * base, 5 * s_b)).sum() >= 2


cells = []
for rho, blend, te in wk.CELLS:
    fails = n = 0
    for u0 in np.linspace(0, 1.995, 160):
        half, ext = te * np.sqrt(4 - u0**2), max(4 * te * 3, 3.0)
        t = np.arange(-half - ext - 0.007, half + ext, wk.CAD)
        f = wk.w3_flux_avg(t, 0.0, te, u0, rho, blend)
        dim = np.flatnonzero(f <= 10**-0.4)
        if len(dim) < 3:
            continue
        n += 1
        i, j = dim[0], dim[-1]
        span = max(t[j] - t[i], 1.0)
        fails += not (ok_side(t, f, i, span, -1) and ok_side(t, f, j, span, 1))
    cells.append(
        {"rho": rho, "blend": blend, "tE": te, "u0_values": n, "fail_share": round(fails / n, 3)}
    )
print(json.dumps({"label": "simulated (model only, no data)", "cells": cells}, indent=1))
