# v11 sample of the chosen area (PLAN.md section 4): stage_sample's logic on the objects table already read by
# obs_script (one download). Usage: python sample_script.py.txt <scratch dir>
import json, sys
import pandas as pd
sys.path.insert(0, 'scripts')
import w3_ztf as wz
SCR = sys.argv[1]
chosen = json.load(open(wz.OUT / 'v11' / 'obs_conditions.json'))['chosen']
assert chosen, 'no area chosen; the plan stops before any light curve'
wz.configure(int(chosen), 'v11')
obj = pd.read_parquet(f'{SCR}/obj959.parquet')
old = [(g['ra'], g['dec']) for g in json.load(open(wz.OUT / 'sample.json'))['groups']]
old += [(g['ra'], g['dec']) for g in json.load(open(wz.OUT / 'v4' / 'sample.json'))['groups']]
groups, info = wz.select(obj, old)
info['calibration_groups'] = int(sum(not g['holdout'] for g in groups))
info['search_groups'] = int(sum(g['holdout'] for g in groups))
info['lc_partition'] = wz.LC_PART
wz.SAMPLE.parent.mkdir(parents=True, exist_ok=True)
wz.SAMPLE.write_text(json.dumps({'label': 'derived; v11 PLAN section 4', 'info': info, 'groups': groups}) + '\n')
n = wz.save_comparisons(obj, {m for g in groups for m in g['members']})
print(json.dumps(info), 'controls', n)
