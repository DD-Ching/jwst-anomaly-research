# v26 dedupe list: the v25 list (v1-v22) plus the six v25 area samples. Positions + objectids.
# Usage: python dedupe_script.py.txt
import gzip, json, os, subprocess
if not os.path.exists('results/w3_ztf/v25/dedupe.json.gz'):
    subprocess.run(['.venv/bin/python', 'results/w3_ztf/v25/dedupe_script.py.txt'], check=True)
ded = json.load(gzip.open('results/w3_ztf/v25/dedupe.json.gz'))['groups']
n0 = len(ded)
for p in (1149, 2345, 2346, 2347, 4988, 4990):
    f = f'results/w3_ztf/v25/{p}/sample.json'
    gs = json.load(open(f) if os.path.exists(f) else gzip.open(f + '.gz'))['groups']
    ded += [[f'v25 ({p})', round(g['ra'], 7), round(g['dec'], 7), g['members']] for g in gs]
out = {'label': 'derived; all earlier sample groups (v1-v25)', 'groups': ded, 'n': len(ded), 'n_before_v25': n0}
gzip.open('results/w3_ztf/v26/dedupe.json.gz', 'wt').write(json.dumps(out, separators=(',', ':')) + '\n')
print(n0, len(ded))
