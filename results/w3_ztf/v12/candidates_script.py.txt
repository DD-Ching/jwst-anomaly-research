# v12 candidate areas from the HATS partition lists only (no data). Usage: python candidates_script.py.txt <lc csv> <obj csv>
import csv, json, sys
import numpy as np
import astropy.units as u
from astropy.coordinates import SkyCoord
from astropy_healpix import HEALPix
lc = [r for r in csv.DictReader(open(sys.argv[1])) if r['Norder'] == '5']
obj4 = {int(r['Npix']) for r in csv.DictReader(open(sys.argv[2])) if r['Norder'] == '4'}
hp = HEALPix(nside=2**5, order='nested')
USED_PARENTS = {959, 1783}  # v1-v5, v11 (Npix 3836-3839) and v6/v10 (Npix 7134)
ok = []
for r in lc:
    pix, n = int(r['Npix']), int(r['num_rows'])
    if pix // 4 not in obj4 or pix // 4 in USED_PARENTS or not 300_000 <= n <= 700_000:
        continue
    lon, lat = hp.healpix_to_lonlat(pix)
    c = SkyCoord(lon.to(u.deg), lat.to(u.deg))
    if 25 <= c.dec.deg <= 65 and abs(c.galactic.b.deg) >= 30:
        ok.append((pix, n, round(c.ra.deg, 2), round(c.dec.deg, 2), round(c.galactic.b.deg, 1)))
parents = sorted({p // 4 for p, *_ in ok})
pick = [parents[i] for i in np.unique(np.linspace(0, len(parents) - 1, 3).round().astype(int))]
cand = []
for k, par in enumerate(pick):
    kids = sorted([x for x in ok if x[0] // 4 == par], key=lambda x: (-x[1], x[0]))[: (2 if k < 2 else 1)]
    cand += kids
cand.append((3837, None, None, None, None))  # fallback: old fields (959), never read; v11 metadata worst cell 0.687
out = {'label': 'derived (HATS partition_info only)', 'rule': 'Norder 5, parent has an objects partition, parent not '
       '959/1783, 300k-700k rows, centre dec +25..+65, |b| >= 30; 3 parents evenly spaced in Npix order; their 2, 2, 1 '
       'largest children; plus Npix 3837 (old fields) as the fallback', 'eligible_partitions': len(ok),
       'eligible_parents': len(parents), 'candidates': [dict(zip(('npix', 'rows', 'ra', 'dec', 'b'), x)) for x in cand]}
json.dump(out, open('results/w3_ztf/v12/candidates.json', 'w'), indent=1); open('results/w3_ztf/v12/candidates.json', 'a').write('\n')
print(json.dumps(out, indent=1))
