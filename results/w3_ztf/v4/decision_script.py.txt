# Frozen pass conditions (PLAN section 3) computed from the v4 JSON outputs. Usage: python decision_script.py.txt
import json
inj = json.load(open('results/w3_ztf/v4/injections_v4.json'))
srch = json.load(open('results/w3_ztf/v4/search_all_v4.json'))['summary']
cells = {(c['rho'], c['blend'], c['tE']): c for c in inj['w3']}
w3 = {f"rho{r}_tE{te:g}": cells[(r, 1.0, te)]['recovered'] / cells[(r, 1.0, te)]['n'] for r in (0.01, 0.1) for te in (30.0, 100.0)}
ctrl = {k: v['passed_vetting'] / v['n'] for k, v in inj['controls'].items()}
cond = {'w3_ge_0.5_tE30_100_blend1': all(v >= 0.5 for v in w3.values()),
        'controls_le_0.10': all(v <= 0.10 for v in ctrl.values()),
        'survivors_le_5': len(srch['passed_groups']) <= 5}
classes = {
    'full_disappearance_rho0.01': {f"blend{b:g}_tE{te:g}": f"{cells[(0.01, b, te)]['recovered']}/{cells[(0.01, b, te)]['n']}" for b in (1.0, 0.7) for te in (3.0, 10.0, 30.0, 100.0)},
    'short_tE3_10': {f"rho{r}_blend{b:g}_tE{te:g}": f"{cells[(r, b, te)]['recovered']}/{cells[(r, b, te)]['n']}" for r in (0.01, 0.1) for b in (1.0, 0.7) for te in (3.0, 10.0)},
    'slow_edge_rho0.1': {f"blend{b:g}_tE{te:g}": f"{cells[(0.1, b, te)]['recovered']}/{cells[(0.1, b, te)]['n']}" for b in (1.0, 0.7) for te in (3.0, 10.0, 30.0, 100.0)},
}
out = {'label': 'derived; frozen rule applied by code', 'w3_pass_cells': w3, 'control_pass_rate': ctrl,
       'survivors': len(srch['passed_groups']), 'conditions': cond, 'all_pass': all(cond.values()),
       'verdict': 'passed' if all(cond.values()) else 'failed (some conditions pass)', 'by_class': classes}
json.dump(out, open('results/w3_ztf/v4/decision.json', 'w'), indent=1); open('results/w3_ztf/v4/decision.json', 'a').write('\n')
print(json.dumps(out, indent=1))
