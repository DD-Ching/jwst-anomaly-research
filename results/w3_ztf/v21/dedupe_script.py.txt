# v21 dedupe list: the v20 list (v1-v16) plus every v20 area sample group (7 started areas, calibration and search,
# including failed areas). Positions + objectids. Usage: python dedupe_script.py.txt
import gzip, json, os, subprocess
if not os.path.exists('results/w3_ztf/v20/dedupe.json.gz'):
    subprocess.run(['.venv/bin/python', 'results/w3_ztf/v20/dedupe_script.py.txt'], check=True)
ded = json.load(gzip.open('results/w3_ztf/v20/dedupe.json.gz'))['groups']
n0 = len(ded)
for p in (1356, 2490, 2491, 8126, 2488, 2489, 2900):
    f = f'results/w3_ztf/v20/{p}/sample.json'
    gs = json.load(open(f) if os.path.exists(f) else gzip.open(f + '.gz'))['groups']
    ded += [[f'v20 ({p})', round(g['ra'], 7), round(g['dec'], 7), g['members']] for g in gs]
out = {'label': 'derived; all earlier sample groups (v1-v20)', 'groups': ded, 'n': len(ded), 'n_before_v20': n0}
gzip.open('results/w3_ztf/v21/dedupe.json.gz', 'wt').write(json.dumps(out, separators=(',', ':')) + '\n')
print(n0, len(ded))
