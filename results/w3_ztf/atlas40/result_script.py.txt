# PLAN section 5: pooled A / B outcomes on the 40 stars, the success condition, Wilson intervals, the development-star
# breakdown and the completion status from tasks.json. Usage: python result_script.py.txt
import json
from math import sqrt

T = {p: json.load(open(f'results/w3_ztf/atlas40/trial_{p}.json')) for p in ('4991', '2344')}
tasks = json.load(open('results/w3_ztf/atlas40/tasks.json'))
targets = json.load(open('results/w3_ztf/atlas40/targets.json'))['targets']
dev = {t['key'] for t in targets if t['dev_461']}
ev = [e for p in T for e in T[p]['events']]


def wilson(k, n, z=1.96):
    if n == 0:
        return None
    p = k / n
    d = 1 + z * z / n
    c = (p + z * z / (2 * n)) / d
    h = z * sqrt(p * (1 - p) / n + z * z / (4 * n * n)) / d
    return [max(0.0, c - h), min(1.0, c + h)]


def success(restored, a_insufficient, fade_pass, fade_n):
    """#459 / ASAS-SN plan section 3: restored / A-insufficient >= 0.50 and fade passes / fade injections <= 0.10."""
    return restored / a_insufficient >= 0.50 and fade_pass / fade_n <= 0.10


def tally(es, b='b'):
    w3 = [e for e in es if e['kind'] != 'fade']
    fd = [e for e in es if e['kind'] == 'fade']
    ins = [e for e in w3 if e['a'] == 'edge_insufficient']
    trans = {}
    for e in es:
        key = f"{'fade' if e['kind'] == 'fade' else 'w3'}: {e['a']} -> {e[b]}"
        trans[key] = trans.get(key, 0) + 1
    d = {'w3_injections': len(w3), 'fade_injections': len(fd), 'w3_a_insufficient': len(ins),
         'w3_restored': sum(e[b] == 'recovered' for e in ins), 'w3_recovered_a': sum(e['a'] == 'recovered' for e in w3),
         'w3_recovered_b': sum(e[b] == 'recovered' for e in w3),
         'w3_lost_in_b': sum(e['a'] == 'recovered' and e[b] != 'recovered' for e in w3),
         'fade_pass_a': sum(e['a'] == 'recovered' for e in fd), 'fade_pass_b': sum(e[b] == 'recovered' for e in fd),
         'fade_insufficient_a': sum(e['a'] == 'edge_insufficient' for e in fd),
         'fade_insufficient_to_slow': sum(e['a'] == 'edge_insufficient' and e[b] == 'edge_slow' for e in fd),
         'transitions': trans}
    d['restored_fraction'] = d['w3_restored'] / d['w3_a_insufficient'] if d['w3_a_insufficient'] else None
    d['restored_wilson95'] = wilson(d['w3_restored'], d['w3_a_insufficient'])
    d['fade_pass_fraction_b'] = d['fade_pass_b'] / d['fade_injections'] if d['fade_injections'] else None
    d['fade_pass_wilson95'] = wilson(d['fade_pass_b'], d['fade_injections'])
    return d


status = {}
for v in tasks.values():
    status[v.get('status')] = status.get(v.get('status'), 0) + 1
missing = {int(k) for k, v in tasks.items() if v.get('status') != 'done'} | {int(t['key']) for t in targets if t['key'] not in tasks}
ended = all(v.get('status') not in ('queued', 'running', 'unfinished_at_deadline', 'timeout') for v in tasks.values()) \
    and len(tasks) == len(targets)
prim = tally(ev)
miss_ins = sum(e['a'] == 'edge_insufficient' and e['kind'] != 'fade' and e['carrier'] in missing for e in ev)
out = {'label': 'derived (trial_<npix>.json, tasks.json); simulated events; ATLAS observed',
       'tasks_status': status, 'stars_with_atlas': sum(v.get('status') == 'done' for v in tasks.values()),
       'stars_without_atlas': sorted(str(k) for k in missing), 'complete': ended,
       'primary_b': prim,
       'success': success(prim['w3_restored'], prim['w3_a_insufficient'], prim['fade_pass_b'], prim['fade_injections']),
       'bounds_if_incomplete': None if ended else {
           'restored_lower': prim['w3_restored'], 'restored_upper': prim['w3_restored'] + miss_ins},
       'secondary_quoted_sigma': tally(ev, 'b_quoted'), 'secondary_keep_9999': tally(ev, 'b_keep9999'),
       'without_dev_stars': tally([e for e in ev if str(e['carrier']) not in dev]),
       'dev_stars_only': tally([e for e in ev if str(e['carrier']) in dev]),
       'per_area': {p: tally(T[p]['events']) for p in T}}
json.dump(out, open('results/w3_ztf/atlas40/result.json', 'w'), indent=1)
open('results/w3_ztf/atlas40/result.json', 'a').write('\n')
print(json.dumps({k: out[k] for k in ('tasks_status', 'stars_with_atlas', 'complete', 'success', 'bounds_if_incomplete')}, indent=1))
print(json.dumps({k: v for k, v in prim.items() if k != 'transitions'}, indent=1))
print(json.dumps(prim['transitions'], indent=1))
for name in ('secondary_quoted_sigma', 'secondary_keep_9999', 'without_dev_stars'):
    d = out[name]
    print(name, d['w3_restored'], '/', d['w3_a_insufficient'], 'fade', d['fade_pass_b'], '/', d['fade_injections'], 'lost', d['w3_lost_in_b'])
