# PLAN section 3: per-gap ATLAS coverage of the ZTF r / g common gaps for the stars with a downloaded result. Observed
# ATLAS photometry only; no injection and no truth. Usage: python analysis_script.py.txt <data dir>
import io, json, os, sys

import numpy as np
import pandas as pd
from astropy import units as u
from astropy.coordinates import EarthLocation, SkyCoord
from astropy.time import Time

D = sys.argv[1]
sel = json.load(open('results/w3_ztf/atlas/selection.json'))
tasks = json.load(open('results/w3_ztf/atlas/tasks.json'))
inv = {(s['area'], s['kind'], s['i'], s['run'], s['side']): s for s in json.load(open('results/w3_ztf/altdata/inventory.json'))['sides']}
PRECISE = 0.10          # sigma of the relative flux
VISIT_GAP = 0.3         # d; exposures closer than this are one visit
BASE_WIN = 60.0         # d before / after the common gap


def read(key):
    txt = open(os.path.join(D, f'atlas_fp_{key}.txt')).read()
    return pd.read_csv(io.StringIO(txt.replace('###', '')), sep=r'\s+')


def quality(df):
    """FAQ (Rest et al.) filters, c and o only."""
    return ((df['duJy'] < 10000) & (df['err'] == 0) & (df['x'] > 100) & (df['x'] < 10460) & (df['y'] > 100)
            & (df['y'] < 10460) & (df['maj'] > 1.6) & (df['maj'] < 5) & (df['min'] > 1.6) & (df['min'] < 5)
            & (df['apfit'] > -1) & (df['apfit'] < -0.1) & (df['mag5sig'] > 17) & (df['Sky'] > 17)
            & df['F'].isin(['c', 'o']))


def hmjd(mjd_start, ra, dec):
    """Mid-exposure (start + 15 s) heliocentric MJD, as the ZTF series times (w3_ztf.exposure_times)."""
    t = Time(np.asarray(mjd_start, float) + 15.0 / 86400.0, format='mjd', scale='utc')
    geo = EarthLocation.from_geocentric(0 * u.m, 0 * u.m, 0 * u.m)
    return (t + t.light_travel_time(SkyCoord(ra * u.deg, dec * u.deg), kind='heliocentric', location=geo)).mjd


def visits(t):
    t = np.sort(np.asarray(t))
    return int(len(t) and 1 + (np.diff(t) > VISIT_GAP).sum())


out = {'label': 'observed (ATLAS forced photometry, reduced mode); derived (counts); gaps from simulated injected events',
       'params': {'precise_sigma_rel': PRECISE, 'visit_gap_d': VISIT_GAP, 'baseline_window_d': BASE_WIN},
       'stars': {}, 'gaps': []}
for key, req in sel['requests'].items():
    if not os.path.exists(os.path.join(D, f'atlas_fp_{key}.txt')):
        out['stars'][key] = {'status': 'no result', 'task': {k: tasks.get(key, {}).get(k) for k in ('task_id', 'starttimestamp', 'finishtimestamp')}}
        continue
    df = read(key)
    q = quality(df).to_numpy()
    df['t'] = hmjd(df['MJD'], req['ra'], req['dec'])
    df['rel'] = np.nan
    df['sig'] = np.nan
    st = {'status': 'result', 'header_ok': open(os.path.join(D, f'atlas_fp_{key}.txt')).read(6) == '###MJD', 'rows': int(len(df)), 'rows_by_filter': df['F'].value_counts().to_dict(),
          'quality_pass': int(q.sum()), 'quality_pass_by_filter': df.loc[q, 'F'].value_counts().to_dict(),
          'mjd_range': [float(df['MJD'].min()), float(df['MJD'].max())] if len(df) else None,
          'negative_flux_rows': int((df['uJy'] < 0).sum()), 'baseline_uJy': {}, 'median_sigma_rel': {}}
    for f in ('c', 'o'):
        m = q & (df['F'] == f).to_numpy()
        if m.sum() < 5:
            continue
        med = float(np.median(df.loc[m, 'uJy']))
        sel_f = (df['F'] == f).to_numpy()
        df.loc[sel_f, 'rel'] = df.loc[sel_f, 'uJy'] / med
        df.loc[sel_f, 'sig'] = df.loc[sel_f, 'duJy'] / med
        st['baseline_uJy'][f] = med
        st['baseline_mag_AB'] = st.get('baseline_mag_AB', {}) | {f: float(-2.5 * np.log10(med) + 23.9)}
        st['median_sigma_rel'][f] = float(np.median(df.loc[m, 'sig']))
        st.setdefault('median_chi_n', {})[f] = float(np.median(df.loc[m, 'chi/N']))
        r = df.loc[m, 'rel'].to_numpy()  # robust scatter of the passing rows vs the quoted sigma (a check of the errors)
        st.setdefault('robust_scatter_rel', {})[f] = float(1.4826 * np.median(np.abs(r - np.median(r))))
        st.setdefault('scatter_over_median_sigma', {})[f] = st['robust_scatter_rel'][f] / st['median_sigma_rel'][f]
    good = q & np.isfinite(df['sig'].to_numpy()) & (df['sig'].to_numpy() <= PRECISE)
    not9999 = (df['uJy'] != 9999).to_numpy()  # review: a suspicious round value passing the cuts (post hoc check only)
    st['quality_pass_precise'] = int(good.sum())
    out['stars'][key] = st
    t = df['t'].to_numpy()
    for g in sel['gaps']:
        if str(g['carrier']) != key:
            continue
        lo, hi = g['common_gap']
        dlo, dhi = g['decisive']
        ing = (t > lo) & (t < hi)
        ind = (t > dlo) & (t <= dhi) if g['side'] == 'post' else (t >= dlo) & (t < dhi)
        before = (t >= lo - BASE_WIN) & (t <= lo)
        after = (t >= hi) & (t <= hi + BASE_WIN)
        rec = {k: g[k] for k in ('area', 'carrier', 'kind', 'i', 'run', 'side', 'common_gap', 'decisive', 'U_J', 'span_j')}
        iv = inv[(g['area'], g['kind'], g['i'], g['run'], g['side'])]
        rec['ztf_r_side_class_v24'] = iv['side_class']  # A no r exposure, B only flagged r exposures, C1 DR24 product gap
        rec['ztf_g_window_459'] = iv['ztf_g']
        rec.update({'gap_days': hi - lo, 'atlas_in_gap': int(ing.sum()), 'atlas_in_gap_quality': int((ing & q).sum()),
                    'atlas_in_gap_quality_precise': int((ing & good).sum()), 'visits_in_gap_quality_precise': visits(t[ing & good]),
                    'atlas_in_decisive': int(ind.sum()), 'atlas_in_decisive_quality_precise': int((ind & good).sum()),
                    'visits_in_decisive_quality_precise': visits(t[ind & good]),
                    'visits_in_decisive_excl_uJy_9999_posthoc': visits(t[ind & good & not9999]),
                    'baseline_visits_before': visits(t[before & good]), 'baseline_visits_after': visits(t[after & good])})
        rec['class'] = ('decisive' if rec['visits_in_decisive_quality_precise'] else
                        'gap only' if rec['visits_in_gap_quality_precise'] else
                        'low quality only' if rec['atlas_in_gap'] else 'none')
        rec['outliers_in_gap'] = [{'hmjd': float(t[k]), 'mjd_start': float(df['MJD'].iloc[k]), **{c: (df[c].iloc[k].item() if hasattr(df[c].iloc[k], 'item') else df[c].iloc[k])
                                   for c in ('uJy', 'duJy', 'F', 'chi/N', 'x', 'y', 'mag5sig', 'Sky', 'Obs')},
                                   'rel': float(df['rel'].iloc[k])}
                                  for k in np.flatnonzero(ing & good) if not 0.759 <= df['rel'].iloc[k] <= 1.3]
        rec['in_gap_epochs'] = [{'t': float(t[k]), 'F': str(df['F'].iloc[k]), 'rel': None if not np.isfinite(df['rel'].iloc[k]) else float(df['rel'].iloc[k]),
                                 'sig': None if not np.isfinite(df['sig'].iloc[k]) else float(df['sig'].iloc[k]), 'quality': bool(q[k])}
                                for k in np.flatnonzero(ing)]
        out['gaps'].append(rec)
cls = {}
for g in out['gaps']:
    cls[g['class']] = cls.get(g['class'], 0) + 1
by_rc = {}
for g in out['gaps']:
    by_rc.setdefault(g['ztf_r_side_class_v24'], {}).setdefault(g['class'], 0)
    by_rc[g['ztf_r_side_class_v24']][g['class']] += 1
q_in_gap = [e['rel'] for g in out['gaps'] for e in g['in_gap_epochs'] if e['quality'] and e['rel'] is not None and e['sig'] is not None and e['sig'] <= PRECISE]
ivals = {}
for g in out['gaps']:
    ivals.setdefault((g['carrier'], round(g['common_gap'][0], 3), round(g['common_gap'][1], 3)), []).append(g)
uniq = {(g['carrier'], e['t'], e['F']) for g in out['gaps'] for e in g['in_gap_epochs']
        if e['quality'] and e['rel'] is not None and e['sig'] is not None and e['sig'] <= PRECISE}
out['summary'] = {'by_ztf_r_side_class': by_rc,
                  'gap_sides': len(out['gaps']), 'distinct_common_gap_intervals': len(ivals),
                  'distinct_intervals_without_atlas': sum(all(g['atlas_in_gap'] == 0 for g in v) for v in ivals.values()),
                  'unique_in_gap_quality_precise_exposures': len(uniq),
                  'decisive_sides_excl_uJy_9999_posthoc': sum(g['visits_in_decisive_excl_uJy_9999_posthoc'] > 0 for g in out['gaps']),
                  'in_gap_quality_precise_rel_flux': {'n': len(q_in_gap), 'min': min(q_in_gap) if q_in_gap else None,
                                                      'median': float(np.median(q_in_gap)) if q_in_gap else None,
                                                      'max': max(q_in_gap) if q_in_gap else None,
                                                      'p01_p99': [float(x) for x in np.percentile(q_in_gap, [1, 99])] if q_in_gap else None},'gaps_analysed': len(out['gaps']), 'gaps_planned': len(sel['gaps']), 'by_class': cls,
                  'stars_with_result': sum(s['status'] == 'result' for s in out['stars'].values())}
json.dump(out, open('results/w3_ztf/atlas/analysis.json', 'w'), indent=1, default=float)
open('results/w3_ztf/atlas/analysis.json', 'a').write('\n')
print(json.dumps(out['summary'], indent=1))
print(json.dumps({k: {kk: v.get(kk) for kk in ('status', 'rows', 'rows_by_filter', 'quality_pass', 'baseline_mag_AB', 'median_sigma_rel',
                                                 'robust_scatter_rel', 'median_chi_n', 'negative_flux_rows')}
                  for k, v in out['stars'].items()}, indent=1, default=float))
for g in out['gaps']:
    print(g['carrier'], g['side'], [round(x, 1) for x in g['common_gap']], g['class'], g['atlas_in_gap'], g['atlas_in_gap_quality_precise'],
          g['visits_in_decisive_quality_precise'], g['baseline_visits_before'], g['baseline_visits_after'])
