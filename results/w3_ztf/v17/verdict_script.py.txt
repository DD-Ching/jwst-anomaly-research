# v17 per-source answers (PLAN section 4) from check.json (planned), followup.json and blend_1625.json (deviations).
# Usage: python verdict_script.py.txt <dir with the three json files> <out json>
import json, sys
import numpy as np
D, OUT = sys.argv[1:3]
chk = {s['npix']: s for s in json.load(open(f'{D}/check.json'))['sources']}
fu = {s['npix']: s for s in json.load(open(f'{D}/followup.json'))['sources']}
bl = json.load(open(f'{D}/blend_1625.json'))


def rng(v):
    v = [x for x in v if x is not None and np.isfinite(x)]
    return [round(float(min(v)), 3), round(float(max(v)), 3)] if v else None


out = {'label': 'derived; post hoc (PLAN.md); deviation results marked', 'sources': {}}
# ---- 1624
c, f = chk[1624], fu[1624]
ce = {e['expid']: e for e in c['exposures']}
nd = [e for e in f['exposures'] if e['nd']]
br = [e for e in f['exposures'] if not e['nd']]
for e in nd:
    ii = e.get('interp')
    if ii:
        b = np.array(ii['bracket_same_offsets_uJy_scaled']); ii['bracket_spread_frac'] = float(np.std(b, ddof=1) / np.median(b))
        ii['sigma_R_i_approx'] = float(ii['R_i'] * ii['bracket_spread_frac'])
col = {x['expid']: x for x in f['column_test']['exposures']}
out['sources']['1624'] = {
    'key': c['key'], 'members': c['members'], 'ra': c['ra'], 'dec': c['dec'], 'filter': c['filter'], 'run_hmjd': c['run'],
    'nd_exposures': len(nd), 'bracket_exposures': len(br),
    'planned_verdict': c['verdict'], 'planned_reason': 'no image-based comparison stars in the 180" cutouts (check.json)',
    'deviation_verdict': ('real dimming evidence' if False else
                          'data problem supported' if all(e['final'] in ('normal', 'data problem supported') for e in nd) else 'undetermined'),
    'nd_final_counts': {k: sum(e['final'] == k for e in nd) for k in ('normal', 'dim', 'undetermined', 'unknown', 'data problem supported')},
    'bracket_R': rng([e['R'] for e in br]), 'sigma_R': rng([e['sigma_R'] for e in f['exposures']]),
    'nd_raw_R': rng([e['R'] for e in nd]), 'nd_R_i': rng([e['interp']['R_i'] for e in nd]),
    'nd_R_i_bracket_spread_frac': rng([e['interp']['bracket_spread_frac'] for e in nd]),
    'nd_bad_pixels_in_aperture': rng([ce[e['expid']]['bad_pixels_in_aperture'] for e in nd]),
    'nd_aperture_pixels': rng([ce[e['expid']]['aperture_pixels'] for e in nd]),
    'bracket_bad_pixels_in_aperture': rng([ce[e['expid']]['bad_pixels_in_aperture'] for e in br]),
    'column_dist_px_nd': rng([col[e['expid']]['target_dist_to_column_px'] for e in nd]),
    'column_dist_px_bracket': rng([col[e['expid']]['target_dist_to_column_px'] for e in br]),
    'column_level_vs_neighbours_DN_nd': rng([ce[e['expid']].get('bad_pixel_level_vs_neighbours_DN') for e in nd]),
    'column_level_vs_neighbours_DN_bracket': rng([ce[e['expid']].get('bad_pixel_level_vs_neighbours_DN') for e in br]),
    'comparison_stars': f['n_comparison_stars'], 'comparison_missing_per_exposure': rng([e['comparison_missing'] for e in f['exposures']]),
    'comparison_dmag_median': rng([e['comparison_dmag_median'] for e in f['exposures']]),
    'psfcat_target_nd': sorted({ce[e['expid']]['psfcat']['status'] for e in nd}),
    'psfcat_target_bracket_dmag': rng([ce[e['expid']]['psfcat'].get('dmag_vs_bracket') for e in br]),
    'psfcat_target_bracket_flags': sorted({ce[e['expid']]['psfcat'].get('flags') for e in br}),
    'centroid_offset_arcsec': rng([ce[e['expid']].get('check2_centroid_offset_arcsec') for e in c['exposures']]),
    'magzp_range': rng([ce[e['expid']]['check1']['magzp'] for e in c['exposures']]),
    'image_type': sorted({ce[e['expid']]['check1']['imgtypcode'] for e in c['exposures']}),
    'maglimit_nd': rng([ce[e['expid']]['maglimit'] for e in nd]), 'seeing_nd': rng([ce[e['expid']]['seeing'] for e in nd]),
    'nd_R_i_by_expid': {str(e['expid']): {'R_i': round(e['interp']['R_i'], 3), 'bracket_spread_frac': round(e['interp']['bracket_spread_frac'], 3), 'final': e['final']} for e in nd},
    'nd_undetermined_expids': [e['expid'] for e in nd if e['final'] != 'data problem supported'],
    'unchecked_v16_nd_now_checked': c['window_undetected_not_checked'],
}
# ---- 1625
c, f = chk[1625], fu[1625]
ce = {e['expid']: e for e in c['exposures']}
nd = [e for e in bl['exposures'] if e['nd']]
br = [e for e in bl['exposures'] if not e['nd']]
fe = {e['expid']: e for e in f['exposures']}
out['sources']['1625'] = {
    'key': c['key'], 'members': c['members'], 'ra': c['ra'], 'dec': c['dec'], 'filter': c['filter'], 'run_hmjd': c['run'],
    'nd_exposures': len(nd), 'bracket_exposures': len(br),
    'planned_verdict': c['verdict'], 'planned_reason': 'one image-based comparison star in the 180" cutouts (check.json)',
    'deviation_verdict': 'undetermined' if any(e['cls'] in ('unknown', 'undetermined') for e in nd) else
                         ('data problem supported' if all(e['cls'] == 'normal' for e in nd) else 'see classes'),
    'nd_classes': {str(e['expid']): e['cls'] for e in nd}, 'nd_reasons': {str(e['expid']): e['reasons'] for e in nd},
    'seeing_nd': rng([e['seeing'] for e in nd]), 'seeing_bracket': rng([e['seeing'] for e in br]),
    'nd_raw_seeing_aperture_R': rng([fe[e['expid']]['R'] for e in nd]),
    'nd_R_net': {str(e['expid']): [round(e['R_net'], 3), round(e['sigma_R'], 3)] for e in nd},
    'bracket_R_net': rng([e['R_net'] for e in br]),
    'neighbour_sep_arcsec': rng([e['neighbour_sep_arcsec'] for e in bl['exposures']]),
    'neighbour_mag': rng([e['neighbour_mag'] for e in bl['exposures']]),
    'psfcat_target_nd': sorted({ce[e['expid']]['psfcat']['status'] for e in nd}),
    'psfcat_target_bracket_dmag': rng([ce[e['expid']]['psfcat'].get('dmag_vs_bracket') for e in br]),
    'psfcat_target_bracket_mag': c['psfcat_bracket_mag_median'],
    'comparison_stars': f['n_comparison_stars'], 'comparison_dmag_median': rng([e['comparison_dmag_median'] for e in f['exposures']]),
    'blend_mechanism': bl['blend_mechanism'],
    'maglimit_nd': rng([ce[e['expid']]['maglimit'] for e in nd]),
}
json.dump(out, open(OUT, 'w'), indent=1); open(OUT, 'a').write('\n')
print(json.dumps(out, indent=1))
