# v17 1625 neighbour test (DEVIATION, post hoc; RESULT.md "Deviations"): followup.json shows a psfcat source of
# r-independent g ~ 14.4 at 6.6" (PA 49 deg) from 709212400017577 (g ~ 18.2). In the 4-4.6" seeing NDs its wing
# enters the seeing-scaled aperture. Its wing at the target is estimated by the same aperture placed at the target's
# position rotated by 90, 180 and 270 deg about the neighbour (PSF wings are near-symmetric); no PSF fit.
# Usage: python blend_script.py.txt <cache dir> <check json> <followup json> <out json>
import json, sys
import numpy as np
from astropy.io import fits
from astropy.table import Table
from astropy.wcs import WCS
sys.path.insert(0, 'scripts')
import w3_ztf as wz
from hit_lightcurve import aperture_photometry
CACHE, CHK, FU, OUT = sys.argv[1:5]
NORMAL_R, DIM_R, NSIG, R_DIM = 0.75, 0.6, 3.0, 10 ** (-0.4)
s = [x for x in json.load(open(CHK))['sources'] if x['npix'] == 1625][0]
fu = {e['expid']: e for e in [x for x in json.load(open(FU))['sources'] if x['npix'] == 1625][0]['exposures']}
ra, dec, key = s['ra'], s['dec'], s['key']
rows = []
for e in s['exposures']:
    h = fits.open(f"{CACHE}/{e['expid']}_{key}_sci.fits")[0]; img = h.data.astype(float); w = WCS(h.header)
    sc = float(np.sqrt(abs(np.linalg.det(w.pixel_scale_matrix)))) * 3600; k = 10 ** (-0.4 * (h.header['MAGZP'] - 23.9))
    t = Table(fits.open(f"{CACHE}/{e['expid']}_psfcat.fits")[1].data)
    d = np.hypot((np.asarray(t['ra']) - ra) * np.cos(np.radians(dec)), np.asarray(t['dec']) - dec) * 3600
    nb = int(np.argmin(np.where((d > 4) & (d < 10), d, np.inf)))  # the 6.6" neighbour
    xt, yt = (float(v) for v in w.world_to_pixel_values(ra, dec))
    xn, yn = (float(v) for v in w.world_to_pixel_values(float(t['ra'][nb]), float(t['dec'][nb])))
    r = e['aperture_r_arcsec']
    ap = lambda x, y: aperture_photometry(img, x, y, r / sc, (2 * r / sc, (2 * r + 5) / sc)) * k
    px, py = w.world_to_pixel_values(np.asarray(t['ra']), np.asarray(t['dec']))
    wing = []
    for ang in (90, 180, 270):
        c, sn = np.cos(np.radians(ang)), np.sin(np.radians(ang))
        dx, dy = xt - xn, yt - yn
        x, y = xn + c * dx - sn * dy, yn + sn * dx + c * dy
        other = np.hypot(px - x, py - y) * sc < max(3.0, r)  # another psfcat source at the rotated position
        wing.append({'angle': ang, 'flux_uJy': float(ap(x, y)), 'other_source': bool(other.any())})
    ww = [v['flux_uJy'] for v in wing if not v['other_source'] and np.isfinite(v['flux_uJy'])]
    W = float(np.median(ww)) if ww else np.nan
    sW = float(np.std(ww, ddof=1)) if len(ww) > 1 else np.nan
    ft = float(ap(xt, yt))
    rows.append({'expid': e['expid'], 'nd': e['nd'], 't': e['t'], 'seeing': e['seeing'], 'aperture_r_arcsec': r,
                 'neighbour_sep_arcsec': float(d[nb]), 'neighbour_mag': float(t['mag'][nb]) + float(fits.open(f"{CACHE}/{e['expid']}_psfcat.fits")[0].header['MAGZP']),
                 'neighbour_minus_target_bracket_mag': float(t['mag'][nb]) + float(fits.open(f"{CACHE}/{e['expid']}_psfcat.fits")[0].header['MAGZP']) - s['psfcat_bracket_mag_median'],
                 'target_aperture_uJy': ft, 'wing_apertures': wing, 'wing_uJy': W, 'wing_spread_uJy': sW,
                 'net_uJy': ft - W, 'noise_uJy': e['noise_uJy'], 'scale': fu[e['expid']]['scale'],
                 'scale_stable': fu[e['expid']]['stable'], 'infobits': e['infobits'], 'bad_bits': e.get('bad_bits_in_aperture', [])})
brk = [r for r in rows if not r['nd']]
nref = float(np.median([r['net_uJy'] / r['scale'] for r in brk]))
bscat = float(np.std([r['net_uJy'] / r['scale'] / nref for r in brk], ddof=1))
for r in rows:
    R = r['net_uJy'] / r['scale'] / nref
    sR = float(np.sqrt(((r['noise_uJy'] ** 2 + np.nan_to_num(r['wing_spread_uJy']) ** 2) ** 0.5 / r['scale'] / nref) ** 2 + bscat ** 2))
    reasons = []
    if r['infobits']: reasons.append(f"infobits {r['infobits']}")
    if not r['scale_stable']: reasons.append('no stable comparison stars')
    if r['bad_bits']: reasons.append(f"bad mask bits {r['bad_bits']}")
    if not np.isfinite(r['wing_spread_uJy']): reasons.append('wing estimate from < 2 positions')
    if not np.isfinite(sR) or (1 - R_DIM) / sR < NSIG: reasons.append('too shallow (1-mag drop < 3 sigma)')
    judge = not reasons; dd = (1 - R) / sR
    cls = 'unknown' if not judge else ('normal' if R >= NORMAL_R and dd < NSIG else 'dim' if R <= DIM_R and dd >= NSIG else 'undetermined')
    r.update(R_net=float(R), sigma_R=sR, judgeable=judge, reasons=reasons, cls=cls)
out = {'label': 'derived from observed sciimg and psfcat (cached by check_script); DEVIATION, post hoc',
       'bracket_net_median_uJy': nref, 'bracket_scatter': bscat, 'exposures': rows}
json.dump(out, open(OUT, 'w'), indent=1); open(OUT, 'a').write('\n')
for r in rows:
    print(r['expid'], r['nd'], r['seeing'], round(r['target_aperture_uJy']), [round(v['flux_uJy']) for v in r['wing_apertures']], [v['other_source'] for v in r['wing_apertures']],
          round(r['net_uJy']), round(r['R_net'], 3), round(r['sigma_R'], 3), r['cls'], r['reasons'], round(r['neighbour_mag'], 3))
# ---- same-exposure test of the blend mechanism (PSF catalogue only): faint stars (target mag +/- 0.5) with a
# >= 2.5 mag brighter psfcat neighbour at 5-8" in the best-seeing bracket catalogue, against isolated faint stars
# (no other source within 15"); per ND exposure the fraction missing (no psfcat source within 1.5"), counted only
# where the bright neighbour (or, for isolated stars, a comparison star) lies inside that exposure's catalogue.
best = min(brk, key=lambda r: r['seeing'])['expid']
tb = Table(fits.open(f"{CACHE}/{best}_psfcat.fits")[1].data); zb = float(fits.open(f"{CACHE}/{best}_psfcat.fits")[0].header['MAGZP'])
mb_ = np.asarray(tb['mag']) + zb
from scipy.spatial import cKDTree
xy = np.c_[np.asarray(tb['ra']) * np.cos(np.radians(dec)), np.asarray(tb['dec'])] * 3600
tree = cKDTree(xy)
s_mag = s['psfcat_bracket_mag_median']
dt = np.hypot((np.asarray(tb['ra']) - ra) * np.cos(np.radians(dec)), np.asarray(tb['dec']) - dec) * 3600
faint = np.nonzero((np.abs(mb_ - s_mag) <= 0.5) & (tb['flags'] == 0) & (dt > 15))[0]  # the target itself excluded
pairs, iso = [], []
for i in faint:
    nn = [j for j in tree.query_ball_point(xy[i], 15) if j != i]
    dd = {j: float(np.hypot(*(xy[j] - xy[i]))) for j in nn}
    br = [j for j in nn if 5 <= dd[j] <= 8 and mb_[j] <= mb_[i] - 2.5 and all(dd[k] > 4 for k in nn if k != j and dd[k] < 5)]
    if br and len([j for j in nn if dd[j] < 5]) == 0:
        pairs.append((i, br[0]))
    elif not nn:
        iso.append(i)
mech = []
for r in rows:
    if not r['nd']:
        continue
    t = Table(fits.open(f"{CACHE}/{r['expid']}_psfcat.fits")[1].data)
    def found(idx):
        return wz.psfcat_match(t['ra'], t['dec'], np.asarray(tb['ra'])[idx], np.asarray(tb['dec'])[idx]) != -1
    pi = np.array([p[0] for p in pairs], int); pb = np.array([p[1] for p in pairs], int)
    inn = found(pb) if len(pb) else np.array([], bool)
    fp = found(pi) if len(pi) else np.array([], bool)
    fi = found(np.array(iso, int)) if iso else np.array([], bool)
    # footprint proxy (review fix): an isolated star counts only if this catalogue has any source within 60"
    near = np.array([np.any(np.hypot((np.asarray(t['ra']) - tb['ra'][i]) * np.cos(np.radians(dec)), np.asarray(t['dec']) - tb['dec'][i]) * 3600 < 60) for i in iso], bool)
    fi = fi[near] if len(fi) else fi
    mech.append({'pairs': [[float(tb['ra'][i]), float(tb['dec'][i]), float(mb_[i]), float(mb_[j]), float(mb_[i] - mb_[j])] for i, j in pairs], 'expid': r['expid'], 'seeing': r['seeing'], 'pairs_in_catalogue': int(inn.sum()),
                 'pair_faint_missing': int((~fp[inn]).sum()), 'isolated_in_footprint': int(len(fi)), 'isolated_missing': int((~fi).sum())})
out['blend_mechanism'] = {'bracket_catalogue': best, 'n_pairs': len(pairs), 'n_isolated': len(iso), 'exposures': mech,
                          'note': 'isolated stars counted only where the exposure catalogue has a source within 60" (footprint proxy)'}
json.dump(out, open(OUT, 'w'), indent=1); open(OUT, 'a').write('\n')
print(out['blend_mechanism'])
