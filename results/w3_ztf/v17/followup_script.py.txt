# v17 follow-up (DEVIATION, post hoc; RESULT.md "Deviations"): the planned image-based comparison stars were too few
# in the 180" cutouts (check.json: 0 for 1624, 1 for 1625), so the same-exposure comparisons use the official psfcat
# of each whole quadrant. No new download: the files cached by check_script.py.txt are reused.
# Usage: python followup_script.py.txt <cache dir> <check json> <out json>
import json, sys
import numpy as np
from astropy.io import fits
from astropy.table import Table
from astropy.wcs import WCS
sys.path.insert(0, 'scripts')
import w3_ztf as wz
from hit_lightcurve import aperture_photometry
CACHE, CHK, OUT = sys.argv[1:4]
chk = json.load(open(CHK))
NORMAL_R, DIM_R, NSIG, R_DIM = 0.75, 0.6, 3.0, 10 ** (-0.4)

BAD_BITS = (2, 8, 9, 10)


def load(eid, key):
    h = fits.open(f'{CACHE}/{eid}_{key}_sci.fits')[0]; m = fits.open(f'{CACHE}/{eid}_{key}_msk.fits')[0]
    w = WCS(h.header); sc = float(np.sqrt(abs(np.linalg.det(w.pixel_scale_matrix)))) * 3600
    return h.data.astype(float), m.data.astype(np.int64), w, sc, 10 ** (-0.4 * (h.header['MAGZP'] - 23.9))


def interp_rows(img, bad):
    """As check_script.py.txt: bad pixels -> mean of the nearest good pixels left and right in the same row."""
    out = img.copy()
    for yy, xx in zip(*np.nonzero(bad)):
        row = bad[yy]; L = xx - 1; Rr = xx + 1
        while L >= 0 and row[L]: L -= 1
        while Rr < len(row) and row[Rr]: Rr += 1
        v = [img[yy, k] for k in (L, Rr) if 0 <= k < len(row)]
        out[yy, xx] = np.mean(v) if v else np.nan
    return out


def interp_flux(eid, key, ra, dec, r, offs):
    img, mk, w, sc, k = load(eid, key)
    x, y = (float(v) for v in w.world_to_pixel_values(ra, dec))
    bb = np.zeros(img.shape, bool)
    for dy, dx in offs:
        yy, xx = int(round(y + dy)), int(round(x + dx))
        if 0 <= yy < bb.shape[0] and 0 <= xx < bb.shape[1]: bb[yy, xx] = True
    return aperture_photometry(interp_rows(img, bb), x, y, r / sc, (2 * r / sc, (2 * r + 5) / sc)) * k


def bad_offsets(eid, key, ra, dec, r):
    img, mk, w, sc, k = load(eid, key)
    x, y = (float(v) for v in w.world_to_pixel_values(ra, dec))
    yy, xx = np.ogrid[:mk.shape[0], :mk.shape[1]]
    ap = (xx - x) ** 2 + (yy - y) ** 2 <= (r / sc) ** 2
    bad = np.zeros(mk.shape, bool)
    for b in BAD_BITS: bad |= ((mk >> b) & 1).astype(bool)
    return np.argwhere(bad & ap) - np.array([y, x])


def cat(eid):
    h = fits.open(f'{CACHE}/{eid}_psfcat.fits'); zp = float(h[0].header['MAGZP']); t = Table(h[1].data)
    t['cmag'] = t['mag'] + zp
    return t


def poly_fit(t):
    """2nd-order (ra, dec) -> (xpos, ypos) of one psfcat quadrant."""
    ra0, de0 = np.median(t['ra']), np.median(t['dec'])
    u = (np.asarray(t['ra']) - ra0) * np.cos(np.radians(de0)); v = np.asarray(t['dec']) - de0
    A = np.c_[np.ones_like(u), u, v, u * u, u * v, v * v]
    cx = np.linalg.lstsq(A, t['xpos'], rcond=None)[0]; cy = np.linalg.lstsq(A, t['ypos'], rcond=None)[0]
    def f(ra, dec):
        uu = (np.asarray(ra) - ra0) * np.cos(np.radians(de0)); vv = np.asarray(dec) - de0
        B = np.c_[np.ones_like(uu), uu, vv, uu * uu, uu * vv, vv * vv]
        return B @ cx, B @ cy
    rx, ry = f(t['ra'], t['dec'])
    return f, float(np.median(np.hypot(rx - t['xpos'], ry - t['ypos'])))


def match(t, ra, dec, r=1.5):
    idx = wz.psfcat_match(t['ra'], t['dec'], ra, dec)
    return idx


out = {'label': 'derived from observed psfcat, sciimg and mskimg already cached by check_script; DEVIATION, post hoc',
       'sources': []}
for s in chk['sources']:
    ex = s['exposures']; ra, dec = s['ra'], s['dec']
    brk = [e['expid'] for e in ex if e['expid'] in s['bracket_exposures']]
    cats = {e['expid']: cat(e['expid']) for e in ex}
    tb = cats[brk[0]]
    tmag = s['psfcat_bracket_mag_median']
    # comparison stars: bracket psfcat sources within 0.5 mag of the target's PSF mag, flags 0, > 15" away,
    # matched with flags 0 in every bracket exposure (PSF magnitudes only)
    sep = np.hypot((np.asarray(tb['ra']) - ra) * np.cos(np.radians(dec)), np.asarray(tb['dec']) - dec) * 3600
    sel = (np.abs(tb['cmag'] - tmag) <= 0.5) & (tb['flags'] == 0) & (sep > 15)
    cra, cde = np.asarray(tb['ra'])[sel], np.asarray(tb['dec'])[sel]
    keep = np.ones(len(cra), bool); bmag = []
    for b in brk:
        j = match(cats[b], cra, cde); ok = j >= 0
        ok[ok] &= np.asarray(cats[b]['flags'])[j[ok]] == 0
        keep &= ok
        bmag.append(np.where(j >= 0, np.asarray(cats[b]['cmag'])[np.maximum(j, 0)], np.nan))
    cra, cde = cra[keep], cde[keep]; bref = np.nanmedian(np.array(bmag)[:, keep], axis=0)
    src = {'npix': s['npix'], 'key': s['key'], 'n_comparison_stars': int(len(cra)),
           'comparison_rule': 'bracket psfcat, |mag - target| <= 0.5, flags 0 in every bracket, > 15" from target',
           'exposures': []}
    tref = s['target_bracket_uJy']['median']
    sc_b = {}
    for b in brk:
        jb = match(cats[b], cra, cde); fb = jb >= 0
        sc_b[b] = 10 ** (-0.4 * float(np.median(np.asarray(cats[b]['cmag'])[jb[fb]] - bref[fb])))
    src['bracket_scales'] = {str(b): v for b, v in sc_b.items()}
    bscat = float(np.std(np.array(s['target_bracket_uJy']['values']) / tref, ddof=1))
    for e in ex:
        t = cats[e['expid']]
        j = match(t, cra, cde); found = j >= 0
        dm = np.asarray(t['cmag'])[j[found]] - bref[found]
        flagged = int((np.asarray(t['flags'])[j[found]] != 0).sum())
        dmc = float(np.median(dm)) if len(dm) else np.nan
        scale = 10 ** (-0.4 * dmc)
        mad = float(1.4826 * np.median(np.abs(10 ** (-0.4 * dm) - scale))) if len(dm) else np.nan
        stable = len(dm) >= 3 and mad / scale <= 0.15
        R = e['flux_uJy'] / scale / tref
        sR = float(np.sqrt((e['noise_uJy'] / scale / tref) ** 2 + bscat ** 2 + (R * mad / scale / np.sqrt(max(len(dm), 1))) ** 2))
        reasons = []
        if e['infobits']: reasons.append(f"infobits {e['infobits']}")
        if not stable: reasons.append('no stable comparison stars')
        if e.get('bad_bits_in_aperture'): reasons.append(f"bad mask bits {e['bad_bits_in_aperture']} in target aperture")
        if (1 - R_DIM) / sR < NSIG: reasons.append('too shallow')
        judge = not reasons
        d = (1 - R) / sR
        cls = 'unknown' if not judge else ('normal' if R >= NORMAL_R and d < NSIG else 'dim' if R <= DIM_R and d >= NSIG else 'undetermined')
        rec = {'expid': e['expid'], 't': e['t'], 'nd': e['nd'], 'comparison_found': int(found.sum()),
               'comparison_missing': int((~found).sum()), 'comparison_flagged': flagged,
               'comparison_dmag_median': dmc, 'scale': scale, 'scale_mad': mad, 'stable': bool(stable),
               'R': float(R), 'sigma_R': sR, 'judgeable': judge, 'reasons': reasons, 'class': cls, 'final': cls}
        # column-interpolated ratio (same rule as check_script, now with the psfcat scale)
        if not judge and e.get('bad_bits_in_aperture') and all(r.startswith('bad mask bits') for r in reasons):
            r = e['aperture_r_arcsec']; off = bad_offsets(e['expid'], s['key'], ra, dec, r)
            fi = interp_flux(e['expid'], s['key'], ra, dec, r, off)
            fib = [interp_flux(b, s['key'], ra, dec, r, off) / sc_b[b] for b in brk]
            Ri = fi / scale / float(np.median(fib))
            rec['interp'] = {'n_pixels_interpolated': int(len(off)), 'flux_uJy': float(fi),
                             'bracket_same_offsets_uJy_scaled': [float(v) for v in fib], 'R_i': float(Ri)}
            ps = e.get('psfcat', {})
            cat_gap = ps.get('status') != 'matched' or ps.get('flags', 0) != 0
            rec['psfcat_gap'] = bool(cat_gap)
            rec['final'] = 'data problem supported' if stable and Ri >= NORMAL_R and cat_gap else 'undetermined'
        src['exposures'].append(rec)
    # ---- 1624: detector-column test with psfcat stars (same exposures)
    if s['npix'] == 1624:
        colres = []
        ref = cats[brk[-1]]  # a bracket catalogue as the star list
        good = (ref['flags'] == 0) & (ref['snr'] >= 20)
        lra, lde = np.asarray(ref['ra'])[good], np.asarray(ref['dec'])[good]
        for e in ex:
            t = cats[e['expid']]
            f, rms = poly_fit(t)
            m = fits.open(f"{CACHE}/{e['expid']}_{s['key']}_msk.fits")[0]
            mk = m.data.astype(np.int64); w = WCS(m.header)
            colmask = ((mk >> 2) & 1).sum(0) >= 0.9 * mk.shape[0]  # LOW RESPONSIVITY over >= 90 % of the cutout rows
            cols = np.nonzero(colmask)[0]
            # psfcat frame offset of the cutout (median over psfcat stars inside the cutout)
            px, py = w.world_to_pixel_values(np.asarray(t['ra']), np.asarray(t['dec']))
            inside = (px >= 0) & (px < mk.shape[1]) & (py >= 0) & (py < mk.shape[0])
            offx = float(np.median(np.asarray(t['xpos'])[inside] - (px[inside] + 1))) if inside.any() else np.nan
            ccols = cols + 1 + offx
            xt, yt = f([ra], [dec])
            xs, ys = f(lra, lde)
            dcol = np.min(np.abs(xs[:, None] - ccols[None, :]), axis=1) if len(ccols) else np.full(len(xs), np.inf)
            jj = match(t, lra, lde); det = (jj >= 0)
            det_clean = det.copy(); det_clean[det] &= np.asarray(t['flags'])[jj[det]] == 0
            on, off = dcol <= 1.0, dcol >= 5.0
            colres.append({'expid': e['expid'], 'nd': e['nd'], 'poly_rms_px': rms, 'bad_columns_psfcat_x': [float(c) for c in ccols],
                           'target_x': float(xt[0]), 'target_dist_to_column_px': float(np.min(np.abs(xt[0] - ccols))) if len(ccols) else None,
                           'on_column_stars': int(on.sum()), 'on_column_detected': int(det[on].sum()), 'on_column_clean': int(det_clean[on].sum()),
                           'off_column_stars': int(off.sum()), 'off_column_detected': int(det[off].sum()), 'off_column_clean': int(det_clean[off].sum())})
        src['column_test'] = {'rule': 'stars of one bracket psfcat (flags 0, snr >= 20) placed by a 2nd-order (ra, dec) -> pixel fit; '
                              'on column: <= 1 px from a LOW RESPONSIVITY column (>= 90 % of cutout rows); off: >= 5 px. '
                              'Assumes the column runs the whole quadrant (seen over the full 178-row cutout only).',
                              'exposures': colres}
    # ---- 1625: neighbours within 15" in each psfcat (blend test, PSF vs PSF)
    if s['npix'] == 1625:
        nb = []
        for e in ex:
            t = cats[e['expid']]
            d = np.hypot((np.asarray(t['ra']) - ra) * np.cos(np.radians(dec)), np.asarray(t['dec']) - dec) * 3600
            k = np.nonzero(d <= 15)[0]
            nb.append({'expid': e['expid'], 'nd': e['nd'], 'seeing': e['seeing'],
                       'sources': [{'sep_arcsec': float(d[i]), 'pa_deg': float(np.degrees(np.arctan2((t['ra'][i] - ra) * np.cos(np.radians(dec)), t['dec'][i] - dec)) % 360),
                                    'mag': float(t['cmag'][i]), 'flux_uJy': float(10 ** (-0.4 * (t['cmag'][i] - 23.9))), 'flags': int(t['flags'][i]),
                                    'sharp': float(t['sharp'][i]), 'chi': float(t['chi'][i])} for i in k[np.argsort(d[k])]]})
        src['neighbours'] = nb
    out['sources'].append(src)
json.dump(out, open(OUT, 'w'), indent=1); open(OUT, 'a').write('\n')
for s in out['sources']:
    print(s['npix'], s['n_comparison_stars'])
    for e in s['exposures']:
        print(' ', e['expid'], e['nd'], e['comparison_found'], e['comparison_missing'], round(e['comparison_dmag_median'], 3), round(e['scale_mad'], 3), round(e['R'], 3), round(e['sigma_R'], 3), e['class'], e['reasons'])
    for c in s.get('column_test', {}).get('exposures', []):
        print('  col', c)
    for n in s.get('neighbours', []):
        print('  nb', n['expid'], n['nd'], n['seeing'], [(round(x['sep_arcsec'], 2), round(x['pa_deg']), round(x['mag'], 2), x['flags']) for x in n['sources']])
