def classify(R, sR):
    d = (1 - R) / sR
    if R >= NORMAL_R and d < NSIG:
        return 'normal'
    if R <= DIM_R and d >= NSIG:
        return 'dim'
    return 'undetermined'


def rstd(v):
    v = np.asarray(v, float); v = v[np.isfinite(v)]
    return float(1.4826 * np.median(np.abs(v - np.median(v)))) if len(v) > 2 else np.nan


class Exp:
    def __init__(self, rec):
        e = rec['expid']; self.rec = rec; self.expid = e
        h = fits.open(f'{D}/{e}_sci.fits')[0]
        self.sci = h.data.astype(float); self.w = WCS(h.header); self.gain = float(h.header['GAIN'])
        self.k = 10 ** (-0.4 * (float(h.header['MAGZP']) - 23.9)); self.magzp = float(h.header['MAGZP'])
        mk = fits.open(f'{D}/{e}_msk.fits')[0].data.astype(np.int64)
        self.bad = np.zeros(mk.shape, bool)
        for b in BAD_BITS:
            self.bad |= ((mk >> b) & 1).astype(bool)
        self.bad |= ~np.isfinite(self.sci)
        self.p = m.read(f'{D}/{e}_psf.psf')
        c = fits.open(f'{D}/{e}_psfcat.fits'); self.cat = Table(c[1].data); self.cat['cmag'] = self.cat['mag'] + float(c[0].header['MAGZP'])
        px, py = self.w.world_to_pixel_values(np.asarray(self.cat['ra']), np.asarray(self.cat['dec']))
        self.cat['px'], self.cat['py'] = px, py
        ins = (px >= 0) & (px < self.sci.shape[1]) & (py >= 0) & (py < self.sci.shape[0])
        # cutout (0-based) -> quadrant (1-based DAOPHOT) offsets from the psfcat stars inside the cutout
        self.offx = float(np.median(np.asarray(self.cat['xpos'])[ins] - (px[ins] + 1)))
        self.offy = float(np.median(np.asarray(self.cat['ypos'])[ins] - (py[ins] + 1)))
        srcm = np.zeros(self.sci.shape, bool)
        yy, xx = np.mgrid[:self.sci.shape[0], :self.sci.shape[1]]
        for x, y in zip(px[ins], py[ins]):
            x0, y0 = int(round(x)), int(round(y))
            srcm[max(y0 - 4, 0):y0 + 5, max(x0 - 4, 0):x0 + 5] = True
        bk = Background2D(self.sci, 50, mask=self.bad | srcm, sigma_clip=SigmaClip(3.0), bkg_estimator=MedianBackground())
        self.sub = self.sci - bk.background
        self.err = np.sqrt(bk.background_rms ** 2 + np.clip(self.sub, 0, None) / self.gain)
        self._models = {}

    def quad(self, x, y):
        return x + 1 + self.offx, y + 1 + self.offy

    def model(self, x, y):
        key = (round(x / 50), round(y / 50))  # PSF varies slowly (linear terms over 3072 px): one model per 50 px
        if key not in self._models:
            self._models[key] = m.image_psf(self.p, *self.quad(x, y), half=11, os=4, analytic_scale=self.p['bright'])
        return self._models[key]

    def pix(self, ra, dec, dra=0.0, ddec=0.0):
        x, y = self.w.world_to_pixel_values(ra + dra, dec + ddec)
        return float(x), float(y)


def fit(ex, img, srcs, box, bad, model_exp=None):
    """Joint PSF fit, positions fixed. srcs: list of (x, y). Returns fluxes (DN), errors, valid PSF fractions."""
    me = model_exp or ex
    mod = me.model(*srcs[0]).copy()
    mod.x_0.fixed = True; mod.y_0.fixed = True
    grp = SourceGrouper(min_separation=2 * box) if len(srcs) > 1 else None
    ph = PSFPhotometry(mod, (box, box), grouper=grp, aperture_radius=3)
    init = Table({'x': [s[0] for s in srcs], 'y': [s[1] for s in srcs], 'flux': [max(float(np.nansum(img[int(round(s[1])) - 2:int(round(s[1])) + 3, int(round(s[0])) - 2:int(round(s[0])) + 3])), 1.0) for s in srcs]})
    try:
        r = ph(img, mask=bad, error=ex.err, init_params=init)
    except Exception as exn:
        return [np.nan] * len(srcs), [np.nan] * len(srcs), [0.0] * len(srcs), str(exn)[:80]
    info = []
    for s in srcs:
        h = box // 2; x0, y0 = int(round(s[0])), int(round(s[1]))
        st = m.stamp(me.p, *me.quad(*s), half=h, analytic_scale=me.p['bright'], frac=(s[0] - x0, s[1] - y0))
        b = bad[y0 - h:y0 + h + 1, x0 - h:x0 + h + 1]
        info.append(float(st[~b].sum() / st.sum()) if b.shape == st.shape else 0.0)
    return [float(v) for v in r['flux_fit']], [float(v) for v in r['flux_err']], info, None


def isolated(ex, lo, hi, avoid=()):
    c = ex.cat; xy = np.c_[c['px'], c['py']]; tr = cKDTree(xy)
    ok = []
    for i in range(len(c)):
        x, y = xy[i]
        if not (lo <= c['cmag'][i] <= hi and c['flags'][i] == 0 and 25 < x < ex.sci.shape[1] - 25 and 25 < y < ex.sci.shape[0] - 25):
            continue
        if len(tr.query_ball_point(xy[i], 15)) > 1 or any(np.hypot(x - ax, y - ay) < 30 for ax, ay in avoid):
            continue
        ok.append(i)
    return ok


def registration(ex, ref):
    """Median (dRA, dDec) of this exposure's psfcat positions against the reference catalogue for bright isolated stars."""
    a = ref.cat; b = ex.cat
    sel = (a['flags'] == 0) & (a['snr'] > 20)
    j = __import__('w3_ztf').psfcat_match(b['ra'], b['dec'], np.asarray(a['ra'])[sel], np.asarray(a['dec'])[sel])
    ok = j >= 0
    return (float(np.median(np.asarray(a['ra'])[sel][ok] - np.asarray(b['ra'])[j[ok]])),
            float(np.median(np.asarray(a['dec'])[sel][ok] - np.asarray(b['dec'])[j[ok]])), int(ok.sum()))

            ff, fe, inf, err = fit(ex, ex.sub, srcs, box, ex.bad)
            fo, _, _, _ = fit(ex, ex.sub, srcs, box, ex.bad, exps[others[e]])
            # residual of the full fitted group (every source subtracted), on a 41x41 region around the target
            x0, y0 = int(round(x)), int(round(y)); H = 20
            Y, X = np.mgrid[y0 - H:y0 + H + 1, x0 - H:x0 + H + 1]
            model_img = np.zeros(X.shape)
            for (sx, sy), f in zip(srcs, ff):
                if np.isfinite(f):
                    md = ex.model(sx, sy).copy(); md.x_0, md.y_0, md.flux = sx, sy, f
                    model_img += md(X, Y)
            resid_full = ex.sub[y0 - H:y0 + H + 1, x0 - H:x0 + H + 1] - model_img
            errw = ex.err[y0 - H:y0 + H + 1, x0 - H:x0 + H + 1]; badw = ex.bad[y0 - H:y0 + H + 1, x0 - H:x0 + H + 1]
            sl = slice(H - h, H + h + 1)
            resid, good = resid_full[sl, sl], ~badw[sl, sl]
            chi = float(np.sum((resid / errw[sl, sl])[good] ** 2) / max(good.sum() - 1, 1))
            ring = None
            if npix == 1625:  # neighbour-wing residual: ring 3-8 px around the neighbour, relative to its fitted peak
                rr = np.hypot(X - srcs[1][0], Y - srcs[1][1]); rmask = (rr >= 3) & (rr <= 8) & ~badw
                mdn = ex.model(*srcs[1]).copy(); mdn.x_0, mdn.y_0, mdn.flux = srcs[1][0], srcs[1][1], ff[1]
                peak = float(mdn(np.array([round(srcs[1][0])]), np.array([round(srcs[1][1])]))[0])
                ring = {'mean_resid_over_peak': float(np.mean(resid_full[rmask]) / peak),
                        'chi2_ring': float(np.sum((resid_full / errw)[rmask] ** 2) / max(rmask.sum() - 1, 1))}
            sc = fu[npix][e]['scale']
            rec = {'nd': bool(tt[e]['nd']), 't': tt[e]['t'], 'seeing': ex.rec['seeing'], 'x': x, 'y': y,
                   'valid_pixels_in_box': int(good.sum()), 'box_pixels': int(good.size), 'info': inf[0],
                   'flux_DN': ff[0], 'flux_err_DN': fe[0], 'flux_uJy': ff[0] * ex.k / sc, 'flux_err_uJy': fe[0] * ex.k / sc,
                   'flux_other_psf_uJy': fo[0] * ex.k / sc, 'reduced_chi2_target_box': chi, 'scale_v17_D1': sc, 'neighbour_ring': ring,
                   'validation_pass': bool(val_ok.get(e)), 'validation_sigma_rel': sig_val.get(e), 'error': err}
            if npix == 1625:
                rec['neighbour_flux_uJy'] = ff[1] * ex.k / sc
            src['exposures'][str(e)] = rec
        fb = np.array([src['exposures'][str(e)]['flux_uJy'] for e in br])
        Fref = float(np.median(fb)); sref = float(np.std(fb, ddof=1))
        src['F_ref_uJy'], src['F_ref_scatter_uJy'] = Fref, sref
        for e in order:
            r = src['exposures'][str(e)]
            sF = np.sqrt(r['flux_err_uJy'] ** 2 + (np.nan_to_num(r['validation_sigma_rel']) * abs(r['flux_uJy'])) ** 2 + (r['flux_uJy'] - r['flux_other_psf_uJy']) ** 2)
            R = r['flux_uJy'] / Fref; sR = float(np.hypot(sF / Fref, R * sref / Fref))
            r.update(R=float(R), sigma_R=sR)
            reasons = []
            if not r['validation_pass']: reasons.append('validation failed for this exposure')
            if r['info'] < MIN_INFO: reasons.append(f"PSF fraction on valid pixels {r['info']:.2f} < {MIN_INFO}")
            if r['error']: reasons.append('fit error')
            r['reasons'] = reasons
            r['class'] = 'unknown' if reasons else classify(R, sR)
            r['excludes_1mag_drop'] = bool(not reasons and (R - R_DIM) / sR >= NSIG)
            r['supports_1mag_drop'] = r['class'] == 'dim'
        lo, hi = chk[npix]['run']
        dim = [e for e in nd if src['exposures'][str(e)]['class'] == 'dim' and lo <= tt[e]['t'] <= hi]
        nights = {int(np.floor(tt[e]['t'] + 0.2)) for e in dim}
        bef = any(src['exposures'][str(e)]['class'] == 'normal' and tt[e]['t'] < lo for e in order)
        aft = any(src['exposures'][str(e)]['class'] == 'normal' and tt[e]['t'] > hi for e in order)
        src['verdict'] = ('real dimming evidence' if len(dim) >= 2 and len(nights) >= 2 and bef and aft else
                          'no dimming in any judgeable ND' if all(src['exposures'][str(e)]['class'] == 'normal' for e in nd) else 'undetermined')
    out['sources'][str(npix)] = src
out['seconds'] = round(time.time() - t0, 1)
v1t = out['sources']['1624'].pop('_v1a_trials', [])
