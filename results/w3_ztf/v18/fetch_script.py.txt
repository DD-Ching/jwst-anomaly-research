# v18 data fetch: for the 20 exposures of the two sources (v17 check.json), the IBE metadata, a 600" science and
# mask cutout, the psfcat, the DAOPHOT PSF file and the centre-PSF image; reader check against the centre image.
# Usage: python fetch_script.py.txt <data dir> <out manifest json>
import importlib.util, io, json, os, sys, time, urllib.parse, urllib.request
import numpy as np
from astropy.io import fits
from astropy.table import Table
D, OUT = sys.argv[1:3]
os.makedirs(D, exist_ok=True)
m = importlib.util.module_from_spec(importlib.util.spec_from_loader('daopsf', loader=None))
exec(open('results/w3_ztf/v18/daopsf.py.txt').read(), m.__dict__)
SIZE, CAP_MB = 600, 190.0
chk = json.load(open('results/w3_ztf/v17/check.json'))
nbytes = {}
t0 = time.time()


def get(url, kind):
    err = None
    for k in range(4):
        try:
            raw = urllib.request.urlopen(url, timeout=180).read(); nbytes[kind] = nbytes.get(kind, 0) + len(raw); return raw
        except Exception as ex:
            err = str(ex)[:80]; time.sleep(5 * (k + 1))
    raise RuntimeError(err)


man = {'label': 'observed (IRSA IBE ZTF science products); download manifest', 'size_arcsec': SIZE, 'exposures': []}
for s in chk['sources']:
    ra, dec, key = s['ra'], s['dec'], s['key']
    ids = [e['expid'] for e in s['exposures']]
    q = urllib.parse.urlencode({'POS': f'{ra},{dec}', 'WHERE': f"expid IN ({','.join(map(str, ids))})", 'ct': 'csv',
                                'COLUMNS': 'expid,filefracday,field,ccdid,qid,filtercode,seeing,maglimit,infobits'})
    meta = Table.read(io.BytesIO(get('https://irsa.ipac.caltech.edu/ibe/search/ztf/products/sci?' + q, 'ibe_search')), format='csv')
    for r in meta:
        if sum(nbytes.values()) / 1e6 > CAP_MB:
            raise SystemExit('download cap reached')
        ffd = str(r['filefracday'])
        base = (f"https://irsa.ipac.caltech.edu/ibe/data/ztf/products/sci/{ffd[:4]}/{ffd[4:8]}/{ffd[8:]}/ztf_{ffd}_"
                f"{int(r['field']):06d}_{r['filtercode']}_c{int(r['ccdid']):02d}_o_q{int(r['qid'])}_")
        files = {}
        for kind, suf, cut in (('sci', 'sciimg.fits', True), ('msk', 'mskimg.fits', True), ('psfcat', 'psfcat.fits', False),
                               ('psf', 'sciimgdao.psf', False), ('psfcent', 'sciimgdaopsfcent.fits', False)):
            fn = f"{D}/{int(r['expid'])}_{kind}{'.fits' if suf.endswith('fits') else '.psf'}"
            url = base + suf + (f'?center={ra},{dec}&size={SIZE}arcsec&gzip=false' if cut else '')
            if not os.path.exists(fn):
                open(fn, 'wb').write(get(url, kind))
            files[kind] = {'path': os.path.basename(fn), 'url': url, 'bytes': os.path.getsize(fn)}
        p = m.read(f"{D}/{int(r['expid'])}_psf.psf")
        cent = fits.open(f"{D}/{int(r['expid'])}_psfcent.fits")[0].data
        v = m.validate_reader(p, cent)[p['bright']]
        man['exposures'].append({'npix': s['npix'], 'key': key, 'expid': int(r['expid']), 'filefracday': ffd,
                                 'field': int(r['field']), 'ccdid': int(r['ccdid']), 'qid': int(r['qid']),
                                 'seeing': float(r['seeing']), 'maglimit': float(r['maglimit']), 'infobits': int(r['infobits']),
                                 'psf_header': {k: p[k] for k in ('npsf', 'nexp', 'psfmag', 'bright', 'xpsf', 'ypsf')},
                                 'psf_par_hwhm_px': [float(x) for x in p['par']],
                                 'reader_max_abs_diff_vs_centre_image': v, 'centre_image_peak': float(cent.max() / cent.sum()),
                                 'files': files})
man['bytes'] = nbytes; man['mb'] = round(sum(nbytes.values()) / 1e6, 3); man['seconds'] = round(time.time() - t0, 1)
json.dump(man, open(OUT, 'w'), indent=1); open(OUT, 'a').write('\n')
print(man['mb'], man['seconds'], max(e['reader_max_abs_diff_vs_centre_image'] for e in man['exposures']), len(man['exposures']))
