# v25 dedupe list: the v22 list (v1-v21) plus the eight v22 area samples. Positions + objectids.
# Usage: python dedupe_script.py.txt
import gzip, json, os, subprocess
if not os.path.exists('results/w3_ztf/v22/dedupe.json.gz'):
    subprocess.run(['.venv/bin/python', 'results/w3_ztf/v22/dedupe_script.py.txt'], check=True)
ded = json.load(gzip.open('results/w3_ztf/v22/dedupe.json.gz'))['groups']
n0 = len(ded)
for p in (1148, 1150, 1151, 1853, 2344, 4989, 4991, 5054):
    f = f'results/w3_ztf/v22/{p}/sample.json'
    gs = json.load(open(f) if os.path.exists(f) else gzip.open(f + '.gz'))['groups']
    ded += [[f'v22 ({p})', round(g['ra'], 7), round(g['dec'], 7), g['members']] for g in gs]
out = {'label': 'derived; all earlier sample groups (v1-v22)', 'groups': ded, 'n': len(ded), 'n_before_v22': n0}
gzip.open('results/w3_ztf/v25/dedupe.json.gz', 'wt').write(json.dumps(out, separators=(',', ':')) + '\n')
print(n0, len(ded))
